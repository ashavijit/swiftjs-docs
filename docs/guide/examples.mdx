## Code Snippets

### Basic HTTP Server

```typescript
import { createApp } from 'swiftjs-core';
const app = createApp();
app.get('/', () => 'Hello World');
app.listen().then(() => console.log('Running on :3000'));
```

### Static File Serving

```typescript
import fs from 'fs';
import path from 'path';

app.get('/public/*', ctx => {
  const filePath = path.join(process.cwd(), ctx.path);
  if (fs.existsSync(filePath)) {
    return fs.readFileSync(filePath);
  }
  ctx.notFound();
});
```

### Custom Middleware

```typescript
app.use(async (ctx, next) => {
  console.log(`Request: ${ctx.method} ${ctx.path}`);
  await next();
});
```

### Error Handling

```typescript
app.addHook('onError', (ctx, err) => {
  ctx.logger.error(err);
  ctx.status(500).json({ error: 'Internal Error' });
});
```
